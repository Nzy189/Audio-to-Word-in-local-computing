```python
import http.client
import json
from urllib.parse import urlsplit

PROMPT = (
    "你是语音转写文稿的整理助手。输入是一段语音识别的原始转写文字，可能包含同音错别字、语气词和无意义重复。\n"
    "请输出严格的 JSON 对象，包含两个字段：\n"
    '"corrected"：优化后的完整原文。\n'
    "- 修正同音字和明显的识别错误\n"
    '- 删除"额""嗯""哼""啊"等无意义语气词\n'
    "- 删除无意义的重复词句\n"
    "- 不改写句式，不增删信息，不改变原意，不改变语言\n"
    "- 保留原文的段落结构\n"
    '"summary"：内容概要。\n'
    "- 用精简的要点列表（bullet points）呈现，每点一行\n"
    "- 只保留核心信息：做了什么、决定了什么、关键数据\n"
    "- 不写引言和套话，不复述原文\n"
    '只输出 JSON，格式：{"corrected": "...", "summary": "..."}，不要输出任何其他内容。'
)


class RefineError(Exception):
    pass


class _Retryable(RefineError):
    pass


def _parse_content(content):
    text = (content or "").strip()
    if text.startswith("```"):
        newline = text.find("\n")
        if newline != -1:
            text = text[newline + 1:]
        if text.rstrip().endswith("```"):
            text = text.rstrip()[:-3]
    text = text.strip()
    try:
        return json.loads(text)
    except ValueError:
        start, end = text.find("{"), text.rfind("}")
        if start != -1 and end > start:
            return json.loads(text[start:end + 1])
        raise


def _request(base_url, model, api_key, user_content, timeout):
    parts = urlsplit(base_url)
    if parts.scheme not in ("http", "https") or not parts.hostname:
        raise RefineError("API 地址格式无效")
    path = (parts.path or "").rstrip("/") + "/chat/completions"
    connection = (http.client.HTTPSConnection if parts.scheme == "https" else http.client.HTTPConnection)(
        parts.hostname, parts.port, timeout=timeout)
    body = json.dumps({
        "model": model,
        "messages": [
            {"role": "system", "content": PROMPT},
            {"role": "user", "content": user_content},
        ],
        "temperature": 0.2,
        "stream": False,
    }).encode()
    headers = {"Content-Type": "application/json", "Authorization": "Bearer " + api_key}
    try:
        connection.request("POST", path, body=body, headers=headers)
        response = connection.getresponse()
        status = response.status
        raw = response.read().decode("utf-8", "replace")
    except (http.client.HTTPException, OSError) as exc:
        raise _Retryable("无法连接 AI 服务：" + str(exc)[:120])
    finally:
        connection.close()
    if status in (401, 403):
        raise RefineError("API Key 无效或没有权限（HTTP " + str(status) + "）")
    if status == 404:
        raise RefineError("API 地址或模型不存在（HTTP 404）")
    if status != 200:
        raise _Retryable("AI 服务返回 HTTP " + str(status))
    return raw


def refine(raw_text, base_url, model, api_key, timeout=120):
    if not (base_url and model and api_key):
        raise RefineError("AI 提炼未配置完整（需要地址、模型和 API Key）")
    if not raw_text or not raw_text.strip():
        raise RefineError("没有可提炼的转写文字")
    user_content = raw_text
    last_error = None
    for attempt in (1, 2):
        try:
            raw = _request(base_url, model, api_key, user_content, timeout)
            payload = json.loads(raw)
            content = payload["choices"][0]["message"]["content"]
            result = _parse_content(content)
            corrected = str(result.get("corrected") or "").strip()
            summary = str(result.get("summary") or "").strip()
            if not corrected and not summary:
                raise _Retryable("AI 返回内容为空")
            return {"corrected": corrected, "summary": summary}
        except _Retryable as exc:
            last_error = str(exc)
        except RefineError:
            raise
        except (ValueError, KeyError, IndexError, TypeError) as exc:
            last_error = "AI 返回格式无法解析：" + str(exc)[:120]
            if attempt == 1:
                user_content = raw_text + "\n\n（请务必只输出 JSON 对象，不要包含任何其他文字。）"
    raise RefineError(last_error or "AI 提炼失败")
```

def llm_settings():
    with db() as conn:
        row = conn.execute("SELECT value FROM settings WHERE key='llm_refine'").fetchone()
    if row:
        config = json.loads(row[0])
        return {
            "enabled": bool(config.get("enabled", True)),
            "base_url": config.get("base_url") or LLM_DEFAULT_URL,
            "model": config.get("model") or LLM_DEFAULT_MODEL,
            "api_key": config.get("api_key", ""),
            "source": "网页设置",
        }
    return {
        "enabled": LLM_ENV_ENABLED,
        "base_url": LLM_ENV_URL or LLM_DEFAULT_URL,
        "model": LLM_ENV_MODEL or LLM_DEFAULT_MODEL,
        "api_key": LLM_ENV_KEY,
        "source": "默认配置",
    }


def claim_refine(job_id):
    with db() as conn:
        claimed = conn.execute(
            "UPDATE jobs SET refine_status='running' WHERE id=? AND (refine_status IS NULL OR refine_status IN ('none','done','failed'))",
            (job_id,))
        return bool(claimed.rowcount)


def run_refine(job_id):
    started = time.time()
    try:
        config = llm_settings()
        with db() as conn:
            row = conn.execute("SELECT text FROM jobs WHERE id=?", (job_id,)).fetchone()
        if row is None or not row["text"]:
            raise llm_refine.RefineError("没有可提炼的转写文字")
        result = llm_refine.refine(row["text"], config["base_url"], config["model"], config["api_key"])
        update(job_id, refine_status="done", refined_text=result["corrected"], summary=result["summary"], refine_error="", refine_at=time.time())
        task_event(job_id, "提炼", "AI 提炼完成，耗时 %.1f 秒" % (time.time() - started))
    except Exception as exc:
        reason = str(exc)[:200] or exc.__class__.__name__
        update(job_id, refine_status="failed", refine_error=reason, refine_at=time.time())
        task_event(job_id, "提炼", "AI 提炼失败：" + reason)


def maybe_refine(job_id):
    config = llm_settings()
    if not config["enabled"] or not config["api_key"]:
        return
    with db() as conn:
        row = conn.execute("SELECT status FROM jobs WHERE id=?", (job_id,)).fetchone()
    if row is None or row["status"] != "done":
        return
    if claim_refine(job_id):
        run_refine(job_id)

@app.post("/api/llm-config")
async def save_llm_config(request: Request):
    try:
        raw = await request.body()
        if len(raw) > 8192:
            raise ValueError()
        body = json.loads(raw)
        current = llm_settings()
        enabled = body.get("enabled", current["enabled"])
        base_url = body.get("base_url") or current["base_url"]
        model = body.get("model") or current["model"]
        api_key = body.get("api_key", "")
        if not isinstance(enabled, bool) or not isinstance(base_url, str) or not isinstance(model, str) or not isinstance(api_key, str):
            raise ValueError()
        if len(base_url) > 512 or len(model) > 128 or len(api_key) > 512:
            raise ValueError()
        base_url = base_url.strip().rstrip("/")
        model = model.strip()
        if not api_key or api_key == current["api_key"][:3] + "***":
            api_key = current["api_key"]
        config = {"enabled": enabled, "base_url": base_url, "model": model, "api_key": api_key}
        with db() as conn:
            conn.execute("INSERT INTO settings(key,value) VALUES ('llm_refine',?) ON CONFLICT(key) DO UPDATE SET value=excluded.value", (json.dumps(config),))
        log.info("AI 提炼设置已保存 | 启用=%s | 模型=%s | 地址=%s | Key=%s", enabled, model or "未填写", base_url or "未填写", "已设置" if api_key else "未设置")
        return llm_config()
    except (ValueError, TypeError, AttributeError):

@app.post("/api/jobs/{job_id}/refine")
def refine_job(job_id: str):
    with db() as conn:
        row = conn.execute("SELECT status FROM jobs WHERE id=?", (job_id,)).fetchone()
    if row is None:
        raise HTTPException(404, "任务不存在或已过期")
    if row["status"] != "done":
        raise HTTPException(409, "转写完成后才能提炼")
    config = llm_settings()

        raise HTTPException(409, "请先在设置中启用 AI 提炼并填写 API Key")

        raise HTTPException(409, "正在提炼中，请稍候")
